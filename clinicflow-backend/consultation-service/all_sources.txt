import org.json.JSONObject;

@Data
@NoArgsConstructor
public class PatientDTO {
    private String id;
    private String mrn;
    private String name;
    private Date dob;
    private String gender;
    private String email;
    private String phoneNumber;
    private String address;

    public PatientDTO(Patient patient) {
        this.id = String.valueOf(patient.getPatientId());
        this.mrn = patient.getMrn();
        this.name = patient.getName();
        this.dob = Date.from(patient.getDob().atStartOfDay(java.time.ZoneId.systemDefault()).toInstant());
        this.gender = patient.getGender();
        System.out.println("Contact Info JSON: " + patient.getContactInfoJson().toString());
        try {
            JSONObject json = new JSONObject(patient.getContactInfoJson());
            this.email = json.getString("email");
            this.phoneNumber = json.getString("phone");
        } catch (JSONException e) {
            // Handle the case where "email" key doesn't exist
            this.email = null;
            this.phoneNumber = null;
        }
        try{
            JSONObject json = new JSONObject(patient.getAddressJson());
            this.address = json.getString("street") != null ? json.getString("street") : "";
            this.address += ", " + (json.getString("city") != null ? json.getString("city") : "");
            this.address += " " + (json.getString("zip") != null ? json.getString("zip") : "");
        } catch (JSONException e) {
            // Handle the case where address JSON is malformed
            this.address = null;
        }
        this.address = patient.getAddressJson();
    }

    public Patient toPatient() {
        Patient patient = new Patient();
        // patient.setMrn(mrn); MRN is generated in service layer
        // patient.setId(id);
        patient.setName(name);
        patient.setDob(dob.toInstant().atZone(java.time.ZoneId.systemDefault()).toLocalDate());
        patient.setGender(gender);
        patient.setContactInfoJson("{\"email\":\"" + email + "\", \"phone\":\"" + phoneNumber + "\"}");
        patient.setAddressJson(address); // Placeholder for address
        patient.setPrimaryContact(email != null ? email : phoneNumber);
        return patient;
    }

@Service
public class PatientAppointmentService {

    private final PatientRepository patientRepository;
    private final AppointmentRepository appointmentRepository;
    private final DoctorRepository doctorRepository;

    public PatientAppointmentService(PatientRepository patientRepository,
                                     AppointmentRepository appointmentRepository,
                                     DoctorRepository doctorRepository) {
        this.patientRepository = patientRepository;
        this.appointmentRepository = appointmentRepository;
        this.doctorRepository = doctorRepository;
    }

    // POST: Register patient
    public Patient registerPatient(Patient patient) {
        if (patient.getMrn() == null || patient.getMrn().isBlank()) {
            patient.setMrn(UUID.randomUUID().toString());
        }
        if (patient.getStatus() == null || patient.getStatus().isBlank()) {
            patient.setStatus("ACTIVE");
        }
        if (patient.getCreatedAt() == null) {
            patient.setCreatedAt(LocalDateTime.now());
        }
        patient.setUpdatedAt(LocalDateTime.now());
        return patientRepository.save(patient);
    }

    // POST: Book appointment
    // FIX: removed false "throws ObjectNotFoundException" â€” never actually thrown here

        appointment.setUpdatedAt(LocalDateTime.now());
        return appointmentRepository.save(appointment);
    }

    // GET: Appointments for patient

        return appointmentRepository.findByPatient(patient);

    // GET: Appointment by ID

                        new ObjectNotFoundException("Patient not found: " + patientId));
    }

        existing.setName(updatedPatient.getName());
